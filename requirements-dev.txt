# Python dependencies required to develop and run the local test/CI gates.
#
# Install with:  python3 -m pip install -r requirements-dev.txt
#
# These mirror what GitHub CI installs (.github/workflows/plugin-tests.yml:
# "Install semgrep + pyyaml"). The local pre-push gate (scripts/ci-local.sh)
# runs the same suites, so the same deps are required on every dev machine.

# Required: parsed by the agent-readiness scanner, the static-analysis adapter,
# the accepted-risks applier, the eval comparative scorer, and the
# security-assessment fixture audit. Without it, bats suites fail mid-run with
# "ModuleNotFoundError: No module named 'yaml'".
PyYAML>=6.0

# Required for the security-assessment suites (SAST). CI installs it alongside
# pyyaml; install locally if you run or modify the security-assessment plugin.
semgrep

# Required by the red-team harness smoke test (tests/security-assessment/harness/
# smoke_test.py) and the harness itself — lib/http_client.py and lib/scoring.py
# do `import httpx`. The numpy/scipy/scikit-learn deps the harness lists in
# plugins/security-assessment/harness/redteam/requirements.txt are NOT needed for
# the smoke test (probes import them lazily inside run()), so they stay out of
# the dev gate set.
httpx>=0.27

# Required by scripts/codebase_recon.py to validate the recon envelope against
# plugins/dev-team/knowledge/schemas/recon-envelope-v1.json before writing.
jsonschema>=4.0

# Required by the Python-hook parity harness (plugins/dev-team/tests/hooks/parity/)
# introduced in #574 as Phase 0 of the bash → python migration (#572). Every
# Python-hook slice from #572 lives or dies by this suite; the harness itself
# is stdlib-only but its test runner is pytest.
pytest>=7.0

# Required by tests/scripts/test_orchestrator.py's @pytest.mark.asyncio tests
# (orchestrator.py's concurrent-dispatch coroutines). Folded into
# chk_hook_units by #677 (epic #668) when tests/scripts/ was wired into CI
# for the first time — without this, those tests fail with "async def
# functions are not natively supported."
pytest-asyncio>=0.23

# Required by the Python static-analysis lane (dev-team plugin): /build's
# self-heal pass and /code-review's static pre-pass dispatch ruff (lint +
# autofix, native SARIF; SARIF needs >= 0.3.1) whenever Python files are in
# scope — including this repo's own Python sources.
ruff>=0.3.1

# Required by the same Python static-analysis lane: mypy (type checking,
# diagnostic-only) feeds /code-review through the Tier 3 JSON adapter, which
# needs the --output json flag introduced in 1.11.
mypy>=1.11
